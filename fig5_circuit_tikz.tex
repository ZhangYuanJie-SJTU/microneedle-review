%% Colors: circ=#2ca02c  intel=#9467bd  sense=#ff7f0e  clin=#d62728

\documentclass[tikz, border=4mm]{standalone}
\usepackage{tikz}
\usetikzlibrary{shapes.geometric, arrows.meta, positioning, fit, backgrounds, calc}

\definecolor{cfab}{HTML}{1f77b4}
\definecolor{csense}{HTML}{ff7f0e}
\definecolor{ccirc}{HTML}{2ca02c}
\definecolor{cintel}{HTML}{9467bd}
\definecolor{cclin}{HTML}{d62728}
\definecolor{cbg}{HTML}{f8f8f8}

\tikzset{
  block/.style = {rectangle, rounded corners=2pt, draw=#1, fill=#1!10,
                  minimum width=2.2cm, minimum height=0.9cm,
                  text centered, font=\small\sffamily, line width=0.8pt},
  smallblock/.style = {rectangle, rounded corners=2pt, draw=#1, fill=#1!15,
                  minimum width=1.8cm, minimum height=0.7cm,
                  text centered, font=\footnotesize\sffamily, line width=0.7pt},
  arr/.style = {-Stealth, thick, color=#1},
  dbl/.style = {Stealth-Stealth, thick, color=#1},
  label/.style = {font=\scriptsize\sffamily, text=#1},
  groupbox/.style = {rectangle, rounded corners=3pt, draw=#1, dashed,
                     fill=#1!5, line width=0.7pt},
}

\begin{document}
\begin{tikzpicture}[node distance=0.45cm and 0.6cm]

% ---- MNA Patch boundary ----
\node[block=csense, text width=2.0cm] (mna) at (0,0)
  {MNA Array\\WE$_1$--WE$_N$\\CE, RE};

% ---- AFE ----
\node[block=ccirc, text width=2.2cm, right=0.7cm of mna] (afe)
  {Analog Front-End\\(AFE)\\3-el. potentiostat\\TIA + MUX + ADC};

% ---- MCU ----
\node[block=cintel, text width=2.4cm, right=0.7cm of afe] (mcu)
  {MCU\\ARM Cortex-M4\\16-bit ADC\\Signal processing\\ML inference};

% ---- BLE/NFC ----
\node[smallblock=ccirc, below=0.55cm of mcu] (ble)
  {Wireless\\BLE 5.0 / NFC};

% ---- Power ----
\node[smallblock=cfab, below=0.55cm of afe] (pwr)
  {Power\\LiPo + PMIC\\(+ TENG/BFC opt.)};

% ---- Smartphone ----
\node[block=cclin, text width=1.8cm, right=0.9cm of mcu] (phone)
  {Smartphone\\App\\Real-time\\display};

% ---- Cloud ----
\node[smallblock=cclin, right=0.6cm of phone] (cloud)
  {Cloud /\\EHR};

% ---- Patch groupbox ----
\begin{scope}[on background layer]
  \node[groupbox=ccirc, fit=(mna)(afe)(mcu)(ble)(pwr),
        inner sep=0.35cm, label={[label=ccirc, anchor=north west]north west:Wearable MNA Patch}] (patch) {};
\end{scope}

% ---- Arrows ----
\draw[arr=csense] (mna) -- node[above, label=csense]{\tiny ISF signal} (afe);
\draw[arr=ccirc]  (afe) -- node[above, label=ccirc]{\tiny digital} (mcu);
\draw[arr=ccirc]  (mcu) -- (ble);
\draw[arr=cfab]   (pwr) -- (afe);
\draw[arr=cfab]   (pwr.east) -- ++(0.3,0) |- (mcu.south);

\draw[dbl=cclin]  (ble.south) -- ++(0,-0.3) -- ++(5.2,0) |- (phone.south)
  node[midway, below, label=cclin]{\tiny BLE / NFC};
\draw[arr=cclin]  (phone) -- node[above, label=cclin]{\tiny relay} (cloud);

% ---- Labels ----
\node[label=csense, below=0.12cm of mna] {\tiny \textcolor{csense}{ISF analytes}};

\end{tikzpicture}
\end{document}
